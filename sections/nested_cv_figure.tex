% Figure 2: one outer fold of the protocol of Section 2.4. The point of the figure is the two orange steps: every
% data-dependent transformation is estimated inside the partition that encloses it, never on the data it is scored on.
% Laid out in three columns so that scaling it to the text width does not make it taller than the page.
\begin{figure}[!htbp]
	\centering
	\resizebox{\textwidth}{!}{%
		\begin{tikzpicture}[
			font=\sffamily\small,
			>=Stealth,
			node distance=5mm and 12mm,
			cvbox/.style={draw=gray!60!black, fill=gray!4, thick, rounded corners=3pt,
				align=center, text width=4.6cm, inner sep=5pt},
			outerstep/.style={cvbox, draw=blue!70!black, fill=blue!5},
			innerstep/.style={cvbox, draw=teal!70!black, fill=teal!5},
			fitstep/.style={cvbox, draw=orange!85!black, fill=orange!9},
			testbox/.style={cvbox, draw=red!65!black, fill=red!4},
			flow/.style={->, thick, draw=gray!65}
			]

			% ---- column A: the outer split ----
			\node[cvbox] (data)
			{\textbf{Data set}\\$N = \NCompounds$ compounds};

			\node[outerstep, below=of data] (split)
			{\textbf{Outer split}\\$\OuterKFolds$ folds $\times\ \OuterRepeats$ repeats\\$=\ \TotalOuterFolds$ evaluations};

			\node[outerstep, below=of split] (pool)
			{\textbf{Outer training pool}\\\footnotesize everything below is learned from it alone};

			% ---- column B: the inner loop ----
			\node[innerstep, right=of data] (inner)
			{\textbf{Inner loop: $\InnerKFolds$ folds}\\\footnotesize $\NArchitectureCombos$ candidates, 1--$\MaxComboSize$ of RF, ET, LGBM, XGB, SVR, kNN};

			\node[fitstep, below=of inner] (prep)
			{\textbf{Preprocessing, fitted here}\\\footnotesize imputation $\rightarrow$ scaling $\rightarrow$ feature selection, on the inner training fold only};

			\node[innerstep, below=of prep] (fit)
			{\textbf{Fit candidate}\\\footnotesize one model, or stacking with a RidgeCV meta-learner};

			\node[innerstep, below=of fit] (score)
			{\textbf{Score the inner validation fold}};

			\node[innerstep, below=of score] (select)
			{\textbf{Select}\\\footnotesize highest mean inner $R^2$; hyperparameters fixed, so only the architecture is chosen};

			% ---- column C: back to the outer loop ----
			\node[testbox, right=of inner] (test)
			{\textbf{Outer test fold}\\\footnotesize held out from every step at the left};

			\node[fitstep, below=of test] (refit)
			{\textbf{Refit on the outer training pool}\\\footnotesize the preprocessing is estimated again, on this pool};

			\node[outerstep, below=of refit] (predict)
			{\textbf{Predict the test fold once}\\\footnotesize clipped to the training range $\pm\,3$ $pIC_{50}$};

			\node[outerstep, below=of predict] (agg)
			{\textbf{Aggregate the \TotalOuterFolds\ folds}\\\footnotesize $R^2$, MAE; Nadeau--Bengio corrected $t$-test, Holm, TOST at $\pm 0.05$};

			\draw[flow] (data) -- (split);
			\draw[flow] (split) -- (pool);
			\draw[flow] (pool.east) -- ++(6mm,0) |- (inner.west);
			\draw[flow] (split.east) -- ++(6mm,0) |- (test.west);
			\draw[flow] (inner) -- (prep);
			\draw[flow] (prep) -- (fit);
			\draw[flow] (fit) -- (score);
			\draw[flow] (score) -- (select);
			\draw[flow] (select.east) -- ++(6mm,0) |- (refit.west);
			\draw[flow] (test) -- (refit);
			\draw[flow] (refit) -- (predict);
			\draw[flow] (predict) -- (agg);

			\begin{scope}[on background layer]
				\node[draw=teal!70!black, fill=teal!3, rounded corners=5pt, inner sep=7pt,
				fit=(inner)(prep)(fit)(score)(select),
				label={[text=teal!60!black, font=\sffamily\bfseries\small]below:Inner loop: architecture selection}] {};
			\end{scope}

		\end{tikzpicture}%
	}
	\caption{\textbf{One outer fold of the evaluation protocol.} The outer loop estimates performance and the inner
		loop selects the architecture, so that no model is scored on the data that chose it: the test fold leaves the
		flow at the split and returns only at the prediction step. The two orange boxes are the reason the estimate is
		leakage-free. Imputation, scaling and supervised feature selection are not computed once on the whole data set
		but estimated again inside each partition they are used in, first on the inner training fold during selection
		and then on the outer training pool for the final fit, and applied to held-out data without re-estimation.
		Because the \TotalOuterFolds\ outer partitions are shared by every representation and every model, all
		comparisons in this study are paired on the same molecules.}
	\label{fig:nested_cv}
\end{figure}
